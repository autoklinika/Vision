cmake_minimum_required(VERSION 3.16)
project(vision_nero_servo_relay)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(control_msgs REQUIRED)
find_package(action_msgs REQUIRED)
find_package(trajectory_msgs REQUIRED)
find_package(moveit_msgs REQUIRED)
add_executable(servo_relay src/servo_relay.cpp)
target_compile_features(servo_relay PRIVATE cxx_std_17)
target_compile_options(servo_relay PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(servo_relay rclcpp control_msgs trajectory_msgs moveit_msgs action_msgs)
install(TARGETS servo_relay DESTINATION lib/vision_nero_servo_relay)
ament_package()
